drop database if exists HIMS;
create database if not exists HIMS;
USE HIMS;
-- Departments Table: 
CREATE TABLE Departments (
    DepartmentID INT PRIMARY KEY AUTO_INCREMENT,
    DepartmentName VARCHAR(100) NOT NULL UNIQUE
);

-- Doctors Table: Stores doctor information
CREATE TABLE Doctors (
    DoctorID INT PRIMARY KEY AUTO_INCREMENT,
    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL,
    Specialty VARCHAR(100),
    DepartmentID INT,
    Email VARCHAR(100) UNIQUE,
    PhoneNumber VARCHAR(20),
    FOREIGN KEY (DepartmentID) REFERENCES Departments(DepartmentID) ON DELETE SET NULL
);
-- Patients Table: Stores patient information
CREATE TABLE Patients (
    PatientID INT PRIMARY KEY AUTO_INCREMENT,
    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL,
    DateOfBirth DATE,
    Gender ENUM('Male', 'Female', 'Other'),
    Address VARCHAR(255),
    PhoneNumber VARCHAR(20) UNIQUE,
    Email VARCHAR(100) UNIQUE
);
-- Appointments Table: Stores appointment details
CREATE TABLE Appointments (
    AppointmentID INT PRIMARY KEY AUTO_INCREMENT,
    PatientID INT,
    DoctorID INT,
    AppointmentDate DATETIME NOT NULL,
    Reason VARCHAR(255),
    Status ENUM('Scheduled', 'Completed', 'Cancelled') DEFAULT 'Scheduled',
    FOREIGN KEY (PatientID) REFERENCES Patients(PatientID) ON DELETE CASCADE,
    FOREIGN KEY (DoctorID) REFERENCES Doctors(DoctorID) ON DELETE CASCADE
);

-- Medical Records Table: Stores medical history 
CREATE TABLE MedicalRecords (
    RecordID INT PRIMARY KEY AUTO_INCREMENT,
    PatientID INT,
    DoctorID INT,
    AppointmentID INT,
    Diagnosis TEXT,
    Treatment TEXT,
    RecordDate DATE NOT NULL,
    FOREIGN KEY (PatientID) REFERENCES Patients(PatientID) ON DELETE CASCADE,
    FOREIGN KEY (DoctorID) REFERENCES Doctors(DoctorID) ON DELETE SET NULL,
    FOREIGN KEY (AppointmentID) REFERENCES Appointments(AppointmentID) ON DELETE SET NULL
);
-- Prescriptions Table: Stores prescription details linked to medical records
CREATE TABLE Prescriptions (
    PrescriptionID INT PRIMARY KEY AUTO_INCREMENT,
    RecordID INT,
    Medication VARCHAR(100) NOT NULL,
    Dosage VARCHAR(50),
    Instructions TEXT,
    FOREIGN KEY (RecordID) REFERENCES MedicalRecords(RecordID) ON DELETE CASCADE
);
-- Bills Table: Stores billing information
CREATE TABLE Bills (
    BillID INT PRIMARY KEY AUTO_INCREMENT,
    PatientID INT,
    AppointmentID INT,
    Amount DECIMAL(10,2) NOT NULL,
    BillDate DATE NOT NULL,
    Paid BOOLEAN DEFAULT FALSE,
    Description VARCHAR(255),
    FOREIGN KEY (PatientID) REFERENCES Patients(PatientID) ON DELETE CASCADE,
    FOREIGN KEY (AppointmentID) REFERENCES Appointments(AppointmentID) ON DELETE SET NULL
);

CREATE INDEX idx_doctor_department ON Doctors(DepartmentID);
CREATE INDEX idx_appointment_patient ON Appointments(PatientID);
CREATE INDEX idx_appointment_doctor ON Appointments(DoctorID);
CREATE INDEX idx_medicalrecord_patient ON MedicalRecords(PatientID);
CREATE INDEX idx_bill_patient ON Bills(PatientID);

-- View all doctors with their department
SELECT d.DoctorID, d.FirstName, d.LastName, d.Specialty, dep.DepartmentName
FROM Doctors d
JOIN Departments dep ON d.DepartmentID = dep.DepartmentID;

-- View all upcoming appointments (after current date: December 24, 2025)
SELECT a.AppointmentID, a.AppointmentDate, a.Reason, a.Status,
       p.FirstName AS PatientFirst, p.LastName AS PatientLast,
       d.FirstName AS DoctorFirst, d.LastName AS DoctorLast
FROM Appointments a
JOIN Patients p ON a.PatientID = p.PatientID
JOIN Doctors d ON a.DoctorID = d.DoctorID
WHERE a.AppointmentDate > '2025-12-24'
ORDER BY a.AppointmentDate;

-- View all appointments for a specific patient (e.g., PatientID = 1: Fatuma Ahmed)
SELECT * FROM Appointments WHERE PatientID = 1;

-- Full history of a patient (e.g., PatientID = 1)
SELECT 
    p.FirstName, p.LastName, p.DateOfBirth, p.Gender, p.PhoneNumber,
    a.AppointmentDate, a.Reason, a.Status,
    d.FirstName AS DoctorFirst, d.LastName AS DoctorLast,
    mr.Diagnosis, mr.Treatment, mr.RecordDate
FROM Patients p
LEFT JOIN Appointments a ON p.PatientID = a.PatientID
LEFT JOIN Doctors d ON a.DoctorID = d.DoctorID
LEFT JOIN MedicalRecords mr ON a.AppointmentID = mr.AppointmentID
WHERE p.PatientID = 1
ORDER BY a.AppointmentDate DESC;

-- All bills for a patient (e.g., PatientID = 1)
SELECT b.BillID, b.Amount, b.BillDate, b.Paid, b.Description
FROM Bills b
WHERE b.PatientID = 1
ORDER BY b.BillDate DESC;

-- Total unpaid bills
SELECT SUM(Amount) AS TotalUnpaid 
FROM Bills 
WHERE Paid = FALSE;

-- Unpaid bills details
SELECT 
    b.BillID, b.Amount, b.BillDate, b.Description,
    p.FirstName, p.LastName, p.PhoneNumber
FROM Bills b
JOIN Patients p ON b.PatientID = p.PatientID
WHERE b.Paid = FALSE
ORDER BY b.BillDate;

-- Revenue summary (paid bills)
SELECT 
    COUNT(*) AS PaidBillsCount,
    SUM(Amount) AS TotalRevenue
FROM Bills 
WHERE Paid = TRUE;

-- Number of appointments per doctor
SELECT 
    d.FirstName, d.LastName, d.Specialty,
    COUNT(a.AppointmentID) AS TotalAppointments
FROM Doctors d
LEFT JOIN Appointments a ON d.DoctorID = a.DoctorID
GROUP BY d.DoctorID
ORDER BY TotalAppointments DESC;

-- Appointments per department
SELECT 
    dep.DepartmentName,
    COUNT(a.AppointmentID) AS AppointmentCount
FROM Departments dep
LEFT JOIN Doctors d ON dep.DepartmentID = d.DepartmentID
LEFT JOIN Appointments a ON d.DoctorID = a.DoctorID
GROUP BY dep.DepartmentID
ORDER BY AppointmentCount DESC;

    a.AppointmentDate

-- All appointments, including if patient record was somehow missing (rare but possible for testing)
SELECT 
    a.AppointmentID, 
    a.AppointmentDate, 
    p.FirstName, p.LastName
FROM Patients p
RIGHT JOIN Appointments a ON p.PatientID = a.PatientID;

-- All patients and all appointments (shows unmatched from both sides)
SELECT p.PatientID, p.FirstName, p.LastName, a.AppointmentID, a.AppointmentDate
FROM Patients p
LEFT JOIN Appointments a ON p.PatientID = a.PatientID

UNION

SELECT p.PatientID, p.FirstName, p.LastName, a.AppointmentID, a.AppointmentDate
FROM Patients p
RIGHT JOIN Appointments a ON p.PatientID = a.PatientID
WHERE p.PatientID IS NULL

ORDER BY PatientID, AppointmentDate;

DROP USER 'hims_accountant'@'localhost';
